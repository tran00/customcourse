FUNCTION calculate_scorm_1_2_score(moodle_id, module_instance, element_id):
    
    # Query to get highest score from attempts
    QUERY = """
        SELECT MAX(score_value) AS highest_score,
               MAX(status_value) AS status,
               MAX(progress_value) AS module_progress
        FROM scorm_attempts
        WHERE user_id = moodle_id
          AND scorm_id = module_instance
          AND element_id IN (element_id, status_id, progress_id)
        GROUP BY attempt_id
        ORDER BY highest_score DESC
        LIMIT 1
    """
    
    data = EXECUTE(QUERY)
    
    IF data EXISTS:
        score = data.highest_score
        status = data.status
        progress = data.module_progress
        
        # Normalize score to 0-100 scale
        IF score IS NOT NULL:
            score_float = CONVERT_TO_FLOAT(score)
            
            IF 0 < score_float <= 1:
                # Score is on 0-1 scale, multiply by 100
                normalized_score = score_float * 100
                
            ELSE IF 1 < score_float <= 100:
                # Score is already on 0-100 scale
                normalized_score = score_float
                
            ELSE IF 100 < score_float <= 1000:
                # Score is on 0-1000 scale, divide by 10
                normalized_score = score_float / 10
                
            ELSE IF 1000 < score_float <= 10000:
                # Score is on 0-10000 scale, divide by 100
                normalized_score = score_float / 100
                
            ELSE IF score_float > 10000:
                # Calculate magnitude and divisor dynamically
                magnitude = LENGTH(INTEGER_PART(score_float)) - 2
                IF magnitude > 0:
                    divisor = 10 ^ magnitude
                    normalized_score = score_float / divisor
                ELSE:
                    normalized_score = 100
        
        RETURN normalized_score
    
    RETURN 0
END FUNCTION



FUNCTION calculate_scorm_1_2_progress(progress_data, status, score):
    
    lastPosition = 0
    progress_processed = FALSE
    
    # Check if completed based on status
    IF status IN ['passed', 'completed', 'complete', 'satisfied']:
        lastPosition = 100
        progress_processed = TRUE
        RETURN lastPosition
    
    # Try parsing progress as numeric value
    TRY:
        progress_float = CONVERT_TO_FLOAT(progress_data)
        IF progress_float <= 1:
            lastPosition = progress_float * 100  # Convert 0-1 to 0-100
        ELSE:
            lastPosition = progress_float
        progress_processed = TRUE
        
    CATCH ValueError:
        # Handle custom formats
        IF "::" IN progress_data OR "_" IN progress_data:
            # Check for completion indicators
            IF "1/1" IN progress_data:
                lastPosition = 100
                status = "completed"
                IF score == 0 OR score IS NULL:
                    score = 100
                progress_processed = TRUE
                
            ELSE IF "c:1" IN progress_data:
                lastPosition = 100
                status = "completed"
                IF score == 0 OR score IS NULL:
                    score = 100
                progress_processed = TRUE
        
        # Try parsing as JSON
        IF NOT progress_processed AND progress_data STARTS_WITH '{':
            json_data = PARSE_JSON(progress_data)
            
            IF "progress" IN json_data:
                prog_val = FLOAT(json_data.progress)
                lastPosition = prog_val * 100 IF prog_val <= 1 ELSE prog_val
                progress_processed = TRUE
                
            ELSE IF json_data.completionStatus == "completed":
                lastPosition = 100
                progress_processed = TRUE
                
            ELSE IF "playerLastPosition" IN json_data:
                userLastPosition = json_data.playerLastPosition.lastPosition
                total_time = GET_TOTAL_TIME(json_data)
                
                IF total_time > 0 AND userLastPosition > 0:
                    lastPosition = (userLastPosition / total_time) * 100
                    progress_processed = TRUE
                
                IF json_data.didReachEnd == TRUE:
                    lastPosition = 100
                    progress_processed = TRUE
        
        # Try Base64 decoding (for compressed suspend_data)
        IF NOT progress_processed:
            suspend_data = PREPARE_BASE64(progress_data)
            decoded = BASE64_DECODE(suspend_data)
            decompressed = ZLIB_DECOMPRESS(decoded)
            json_data = PARSE_JSON(decompressed)
            
            IF "playerLastPosition" IN json_data:
                userLastPosition = json_data.playerLastPosition.lastPosition
                chapter_id = json_data.playerLastPosition.chapterId
                total_time = json_data.chapters[chapter_id].end
                
                lastPosition = (userLastPosition / total_time) * 100
                progress_processed = TRUE
                
                IF json_data.didReachEnd == TRUE:
                    lastPosition = 100
    
    # Special case: if score is 100 but progress is 0
    IF (score == 100 OR score > 99) AND lastPosition == 0 AND NOT progress_processed:
        lastPosition = 100
        progress_processed = TRUE
    
    # Cap progress between 0 and 100
    lastPosition = MAX(0, MIN(100, lastPosition))
    
    # Override based on status
    IF status == "passed" OR score == 100:
        lastPosition = 100
    
    RETURN lastPosition
END FUNCTION
